%==== 块高斯消去 ====
\begin{align}
\widetilde{K}&=U^T K U=\operatorname{diag}\left(K_{11}, K_{22}, \widetilde{K}_{33}, K_{44}, K_{55}, \widetilde{K}_{66}, \widetilde{K}_{77}\right) \\
\widetilde{M}&=U^T M U \\
U&=\prod_{i=1}^{6} U_{i} \\
U_{i} &= \left( \begin{array}{c c c c c} I & & & & \\ & \ddots & \Psi_{i j} & & \Psi_{i k} \\ & & I & & \\ & & & \ddots & \\ & & & & I \end{array} \right),
\end{align}

%叶子节点i
\begin{align}
\Psi_{i j} &= -K_{ii}^{-1}K_{ij} \\
\widetilde{K}_{ij}^{[i]} &= 0 \\
\widetilde{M}_{ij}^{[i]} &= M_{ij}+M_{ii}\Psi_{ij} \\
\widetilde{K}_{jk}^{[i]} &= \widetilde{K}_{jk}^{[i-1]}+\Psi_{i j}^T K_{ik} \\
\widetilde{M}_{jk}^{[i]} &= \widetilde{M}_{jk}^{[i-1]}+\Psi_{i j}^T M_{ik}+M_{ij}^T\Psi_{i k}+\Psi_{i j}^T M_{ii}\Psi_{i k}
\end{align}

%非叶子节点i
\begin{align}
\Psi_{i j} &= -\big(\widetilde{K}_{ii}^{[i-1]}\big)^{-1}\widetilde{K}_{ij}^{[i-1]} \\
\widetilde{K}_{ij}^{[i]} &= 0 \\
\widetilde{M}_{ij}^{[i]} &= \widetilde{M}_{ij}^{[i-1]}+\widetilde{M}_{ii}^{[i-1]}\Psi_{ij} \\
\widetilde{K}_{jk}^{[i]} &= \widetilde{K}_{jk}^{[i-1]}+\Psi_{i j}^T \widetilde{K}_{ik}^{[i-1]} \\
\widetilde{M}_{jk}^{[i]} &= \widetilde{M}_{jk}^{[i-1]}+\Psi_{i j}^T \widetilde{M}_{ik}^{[i-1]}+\big(\widetilde{M}_{ij}^{[i-1]}\big)^T\Psi_{i k}+\Psi_{i j}^T\widetilde{M}_{ii}^{[i-1]}\Psi_{i k}\\
(\widetilde{K}_{ii},\widetilde{M}_{ii})&=\big(\widetilde{K}_{ii}^{[i-1]},\widetilde{M}_{ii}^{[i-1]}\big)
\end{align}
